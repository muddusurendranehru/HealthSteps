CREATE OR REPLACE FUNCTION signup_user(p_email text, p_password text)
RETURNS jsonb AS $$
BEGIN
  INSERT INTO users (email, password) VALUES (p_email, p_password);
  RETURN jsonb_build_object('success', true, 'email', p_email);
EXCEPTION WHEN unique_violation THEN
  RETURN jsonb_build_object('success', false, 'message', 'Email exists');
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION login_user(p_email text, p_password text)
RETURNS jsonb AS $$
DECLARE
  user_record RECORD;
BEGIN
  SELECT id, email INTO user_record FROM users 
  WHERE email = p_email AND password = p_password;
  
  IF FOUND THEN
    RETURN jsonb_build_object('success', true, 'user', jsonb_build_object('id', user_record.id, 'email', user_record.email));
  ELSE
    RETURN jsonb_build_object('success', false, 'message', 'Invalid credentials');
  END IF;
END;
$$ LANGUAGE plpgsql;

  const email = sessionStorage.getItem('user_email');
  if (!email) {
    router.push('/signup');
    return null;
  }
  
  // Verify user exists in database
  const { data } = await supabase.rpc('verify_user', { p_email: email });
  if (!data?.exists) {
    sessionStorage.removeItem('user_email');
    router.push('/signup');
    return null;
  }
